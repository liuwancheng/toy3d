add_library(Toy3dProcess STATIC)
target_sources(Toy3dProcess PRIVATE "process.h" "process.cpp")
target_compile_features(Toy3dProcess PUBLIC cxx_std_17)
set_target_properties(Toy3dProcess PROPERTIES CXX_EXTENSIONS OFF FOLDER "Engine/Core")
target_include_directories(Toy3dProcess PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/..>")
target_link_libraries(Toy3dProcess PUBLIC Toy3dFileSystem PRIVATE Toy3dText)

    target_compile_options(Toy3dProcess PRIVATE /utf-8)
endif()

        if(MSVC)
            target_compile_options(${TOY3D_PROCESS_TEST_TARGET} PRIVATE /utf-8)
        endif()

    add_test(NAME Toy3dCore.Process COMMAND Toy3dProcessTests "$<TARGET_FILE:Toy3dProcessHelper>")
