set(TOY3D_SHADER_MAP_OUTPUT_DIR "${CMAKE_BINARY_DIR}/generated/shader_map")
set(TOY3D_SHADER_PARAMETERS_OUTPUT_DIR "${CMAKE_BINARY_DIR}/generated/shader_parameters")
# directory | logical name | relative source | pass | usage; shared with Editor.
set(TOY3D_BUILTIN_SHADER_SOURCES
    "unlit|Toy3d/Surface/Unlit|surface/unlit.shader|Forward|Material"
    "phong|Toy3d/Surface/Phong|surface/phong.shader|Forward|Material"
    "tonemap|Toy3d/PostProcess/Tonemap|postprocess/tonemap.shader|Tonemap|Global"
    "imgui|Toy3d/UI/ImGui|ui/imgui.shader|ImGui|Global"
    "hit_proxy|Toy3d/Editor/HitProxy|editor/hit_proxy.shader|HitProxy|MeshPass"
    "shadow|Toy3d/ShadowDepth/Default|shadow_depth/default.shader|ShadowDepth|MeshPass")
set(TOY3D_SHADER_PARAMETERS_HEADERS "${TOY3D_SHADER_PARAMETERS_OUTPUT_DIR}/builtin_shader_parameters.generated.h")
set(TOY3D_SHADER_SOURCE_FILES)
set(TOY3D_SHADER_STAMPS)
set(TOY3D_BUILTIN_SHADER_RECORDS "")
list(LENGTH TOY3D_BUILTIN_SHADER_SOURCES TOY3D_BUILTIN_SHADER_COUNT)
# Conservative include invalidation for the small controlled builtin tree.
# Exact dependency hashes remain verified in compiler manifests.
file(GLOB_RECURSE TOY3D_SHADER_INCLUDE_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*")
foreach(TOY3D_SHADER_RECORD IN LISTS TOY3D_BUILTIN_SHADER_SOURCES)
    string(REPLACE "|" ";" TOY3D_SHADER_FIELDS "${TOY3D_SHADER_RECORD}")
    list(GET TOY3D_SHADER_FIELDS 0 TOY3D_SHADER_DIRECTORY)
    list(GET TOY3D_SHADER_FIELDS 1 TOY3D_SHADER_NAME)
    list(GET TOY3D_SHADER_FIELDS 2 TOY3D_SHADER_RELATIVE)
    list(GET TOY3D_SHADER_FIELDS 3 TOY3D_SHADER_PASS)
    list(GET TOY3D_SHADER_FIELDS 4 TOY3D_SHADER_USAGE)
    set(TOY3D_SHADER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/builtin/${TOY3D_SHADER_RELATIVE}")
    set(TOY3D_SHADER_OUTPUT "${TOY3D_SHADER_MAP_OUTPUT_DIR}/${TOY3D_SHADER_DIRECTORY}")
    set(TOY3D_SHADER_STAMP "${TOY3D_SHADER_MAP_OUTPUT_DIR}/${TOY3D_SHADER_DIRECTORY}.stamp")
    string(TOLOWER "${TOY3D_SHADER_NAME}" TOY3D_SHADER_HEADER)
    string(REPLACE "/" "_" TOY3D_SHADER_HEADER "${TOY3D_SHADER_HEADER}")
    list(APPEND TOY3D_SHADER_PARAMETERS_HEADERS "${TOY3D_SHADER_PARAMETERS_OUTPUT_DIR}/${TOY3D_SHADER_HEADER}.generated.h")
    list(APPEND TOY3D_SHADER_SOURCE_FILES "${TOY3D_SHADER_SOURCE}")
    list(APPEND TOY3D_SHADER_STAMPS "${TOY3D_SHADER_STAMP}")
    string(APPEND TOY3D_BUILTIN_SHADER_RECORDS
        "        {\"${TOY3D_SHADER_NAME}\", \"${TOY3D_SHADER_RELATIVE}\", \"${TOY3D_SHADER_PASS}\", \"${TOY3D_SHADER_DIRECTORY}\", BuiltinShaderUsage::${TOY3D_SHADER_USAGE}},\n")
    add_custom_command(OUTPUT "${TOY3D_SHADER_STAMP}"
        COMMAND ${CMAKE_COMMAND} -E remove_directory "${TOY3D_SHADER_OUTPUT}"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${TOY3D_SHADER_OUTPUT}"
        COMMAND $<TARGET_FILE:Toy3dShaderCompiler> --toolchain-root "${TOY3D_SHADER_TOOLCHAIN_ROOT}" compile-vulkan
            "${TOY3D_SHADER_SOURCE}" "/Engine/Shaders/${TOY3D_SHADER_RELATIVE}" "${TOY3D_SHADER_PASS}"
            "${TOY3D_SHADER_OUTPUT}" "${CMAKE_BINARY_DIR}/generated/shader_work/${TOY3D_SHADER_DIRECTORY}"
            --engine-include-root "${CMAKE_CURRENT_SOURCE_DIR}/include"
        COMMAND ${CMAKE_COMMAND} -E remove_directory "${BINARY_ROOT_DIR}/shader/${TOY3D_SHADER_DIRECTORY}"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${BINARY_ROOT_DIR}/shader"
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${TOY3D_SHADER_OUTPUT}" "${BINARY_ROOT_DIR}/shader/${TOY3D_SHADER_DIRECTORY}"
        COMMAND ${CMAKE_COMMAND} -E touch "${TOY3D_SHADER_STAMP}"
        DEPENDS Toy3dShaderCompiler "${TOY3D_SHADER_SOURCE}" ${TOY3D_SHADER_INCLUDE_FILES} VERBATIM)
endforeach()
configure_file("builtin_shader_sources.h.in" "${TOY3D_SHADER_PARAMETERS_OUTPUT_DIR}/builtin_shader_sources.generated.h" @ONLY)
set(TOY3D_SHADER_PARAMETERS_STAMP "${TOY3D_SHADER_PARAMETERS_OUTPUT_DIR}/generation.stamp")
add_custom_command(OUTPUT "${TOY3D_SHADER_PARAMETERS_STAMP}"
    BYPRODUCTS ${TOY3D_SHADER_PARAMETERS_HEADERS}
        "${TOY3D_SHADER_PARAMETERS_OUTPUT_DIR}/shader_parameters.outputs"
        "${TOY3D_SHADER_PARAMETERS_OUTPUT_DIR}/shader_parameters.dependencies"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${TOY3D_SHADER_PARAMETERS_OUTPUT_DIR}"
    COMMAND $<TARGET_FILE:Toy3dShaderCompiler> generate-parameters "${TOY3D_SHADER_PARAMETERS_OUTPUT_DIR}" ${TOY3D_SHADER_SOURCE_FILES}
    COMMAND ${CMAKE_COMMAND} -E touch "${TOY3D_SHADER_PARAMETERS_STAMP}"
    DEPENDS Toy3dShaderCompiler ${TOY3D_SHADER_SOURCE_FILES} ${TOY3D_SHADER_INCLUDE_FILES} VERBATIM)
set_source_files_properties(${TOY3D_SHADER_PARAMETERS_HEADERS} PROPERTIES GENERATED TRUE)
add_custom_target(Toy3dShaderParameters DEPENDS "${TOY3D_SHADER_PARAMETERS_STAMP}")
set_target_properties(Toy3dShaderParameters PROPERTIES FOLDER "Engine/Shader")
set(TOY3D_OUTPUT_SHADER_DEPLOY_STAMP "${TOY3D_SHADER_MAP_OUTPUT_DIR}/output_deploy.stamp")
set(TOY3D_OUTPUT_SHADER_DEPLOY_ROOT "${BINARY_ROOT_DIR}/shader/output")
set(TOY3D_OUTPUT_SHADER_COPY_COMMANDS)
foreach(TOY3D_SHADER_RECORD IN LISTS TOY3D_BUILTIN_SHADER_SOURCES)
    string(REPLACE "|" ";" TOY3D_SHADER_FIELDS "${TOY3D_SHADER_RECORD}")
    list(GET TOY3D_SHADER_FIELDS 0 TOY3D_SHADER_DIRECTORY)
    list(GET TOY3D_SHADER_FIELDS 4 TOY3D_SHADER_USAGE)
    if(NOT TOY3D_SHADER_USAGE STREQUAL "Material")
        list(APPEND TOY3D_OUTPUT_SHADER_COPY_COMMANDS COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${TOY3D_SHADER_MAP_OUTPUT_DIR}/${TOY3D_SHADER_DIRECTORY}" "${TOY3D_OUTPUT_SHADER_DEPLOY_ROOT}")
    endif()
endforeach()
add_custom_command(OUTPUT "${TOY3D_OUTPUT_SHADER_DEPLOY_STAMP}"
    COMMAND ${CMAKE_COMMAND} -E remove_directory "${TOY3D_OUTPUT_SHADER_DEPLOY_ROOT}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${TOY3D_OUTPUT_SHADER_DEPLOY_ROOT}"
    ${TOY3D_OUTPUT_SHADER_COPY_COMMANDS}
    COMMAND ${CMAKE_COMMAND} -E touch "${TOY3D_OUTPUT_SHADER_DEPLOY_STAMP}"
    DEPENDS ${TOY3D_SHADER_STAMPS} VERBATIM)
add_custom_target(Toy3dShaders DEPENDS ${TOY3D_SHADER_STAMPS} "${TOY3D_OUTPUT_SHADER_DEPLOY_STAMP}")
set_target_properties(Toy3dShaders PROPERTIES FOLDER "Engine")
set(TOY3D_SHADER_MAP_OUTPUT_DIR "${TOY3D_SHADER_MAP_OUTPUT_DIR}" PARENT_SCOPE)
set(TOY3D_SHADER_PARAMETERS_OUTPUT_DIR "${TOY3D_SHADER_PARAMETERS_OUTPUT_DIR}" PARENT_SCOPE)
set(TOY3D_SHADER_PARAMETERS_HEADERS "${TOY3D_SHADER_PARAMETERS_HEADERS}" PARENT_SCOPE)
